         p_len>=SL_MIN_ZERO && p_len<=SL_MAX_ZERO*/) b=0;
      else
      {
        if(starline.dat_bit>=SL_MIN_BITS) starline.state=100;
          else starline.state=0;
        break;
      }


      if(b) setbit(starline.data, starline.dat_bit);
      if(++starline.dat_bit==SL_MAX_BITS) starline.state=100;
      break;
  }

/////////////// A piece from the Wiki on encoding and decoding Keeloq functions
# define KeeLoq_NLF 0x3A5C742E
# define bitn(x,n) (((x)>>(n))&1)
# define g5(x,a,b,c,d,e) (bitn(x,a)+bitn(x,b)*2+bitn(x,c)*4+bitn(x,d)*8+bit n(x,e)*16)


uint32_t KeeLoq_Encrypt (const uint32_t data, const uint64_t key){
uint32_t x = data, r;
for (r = 0; r < 528; r++)
x = (x>>1)^((bitn(x,0)^bitn(x,16)^(u32)bitn(key,r&63)^ bitn(KeeLoq_NLF,g5(x,1,9,20,26,31)))<<31);
return x;
}


uint32_t KeeLoq_Decrypt (const uint32_t data, const uint64_t key){
uint32_t x = data, r;
for (r = 0; r < 528; r++)
x = (x<<1)^bitn(x,31)^bitn(x,15)^(u32)bitn(key,(15-r)&63)^bitn(KeeLoq_NLF,g5(x,0,8,19,25,30));
return x;
}


boolean CheckValue(unsigned int base, unsigned int value) {
return ((value == base) || ((value > base) && ((value - base) < MAX_DELTA)) || ((value < base) && ((base - value) < MAX_DELTA)));
}

// This function is called by an interrupt on the pin where the receiver is located
void rx_int()
{
noInterrupts();
len=micros()-last;
last=micros();
interrupts();
if(digitalRead(rxPin)==HIGH)
{
level=1;
lolen=len;
}
else
{
level=0;
hilen=len;
}


/////////////////// Receiving Doorhan and similar
if (level == 1) {
if (CheckValue(Pe, hilen) && CheckValue(Pe2, lolen)) {
if (bcounter < 32)
code1 = (code1 << 1) | 1;
else if (bcounter < 64)
code2 = (code2 << 1) | 1;
bcounter++;
}
else if (CheckValue(Pe2, hilen) && CheckValue(Pe, lolen)) { // valid 0
if (bcounter < 32)
code1 = (code1 << 1) | 0;
else if (bcounter < 64)
code2 = (code2 << 1) | 0; bcounter++;
}
else
bcounter = 0;
// Caught 64 bits, which means we write them into variables for further work with them (half the work is done!)
if (bcounter >= 65) {
Cache1 = code1;// first piece of code (encrypted)
Cache2 = code2;// second piece of code (open)
bcounter = 0;


r=1;// This means that we caught the key and can work with it
code1 = 0;
code2 = 0;
}
}
}


////////////////// Function for mirroring the code, for further decoding.
long inver(long Cache345)
{
long Cache2RR;
for (int i = 0; i < 32; i++)
{
bitWrite(Cache2RR,i,bitRead(Cache345,31-i));
}
return Cache2RR;
}

// Function that sends packets like DoorHan and similar
void SendANMotors(long c1, long c2)
{
noInterrupts(); // temporarily disable packet capture to avoid capturing our own transmissions
for (int j = 0; j < 4; j++)
{
// sending 12 initial 0-1 pulses
for (int i = 0; i < 12; i++) {
delayMicroseconds(Pe);
digitalWrite(txPin, HIGH);
delayMicroseconds(Pe);
digitalWrite(txPin, LOW);
}
delayMicroseconds(Pe * 10);
// sending the first half
for (int i = 4 * 8; i > 0; i--) {
SendBit(bitRead(c1, i - 1));
}
// the second half
for (int i = 4 * 8; i > 0; i--) {
SendBit(bitRead(c2, i - 1));
}
// and a couple of unnecessary bits, which mean the battery and repeat flag
SendBit(1);
SendBit(1);
delayMicroseconds(Pe * 39);
}
interrupts();
}
// bit by bit transmission
void SendBit(byte b) {
if (b == 0) {
digitalWrite(txPin, HIGH); // 0
delayMicroseconds(Pe2);
digitalWrite(txPin, LOW);
delayMicroseconds(Pe);
}
else {
digitalWrite(txPin, HIGH); // 1
delayMicroseconds(Pe);
digitalWrite(txPin, LOW);
delayMicroseconds(Pe2);
}
}
